    #!/bin/bash

#log_file="/var/log/nfs_mount_checker.log"
date_now=$(date '+%Y-%m-%d %H:%M:%S')

echo "[$date_now] 自动扫描 fstab 中的网络挂载点..."

# 读取 /etc/fstab 中类型为 nfs/nfs4/cifs/smbfs 的挂载点
mount_points=($(awk '$1 !~ /^#/ && ($3 ~ /^(nfs4?|cifs|smbfs)$/ || $4 ~ /_netdev/){print $2}' /etc/fstab))

for mount_point in "${mount_points[@]}"; do
    if mountpoint -q "$mount_point"; then
        echo "[$date_now] $mount_point 已挂载。"
    else
        echo "[$date_now] $mount_point 未挂载，尝试重新挂载..."
        # 超过 10 秒没有挂载成功则取消挂载命令，防止脚本卡住
        timeout 10 mount "$mount_point"
        if [ $? -eq 0 ]; then
            echo "[$date_now] $mount_point 重新挂载成功。"
        else
            echo "[$date_now] $mount_point 重新挂载失败或超时。"
        fi
    fi
done

自签名 SSL 证书

生成CA机构根私钥和根证书

生成根私钥

1

openssl genpkey -algorithm RSA -out rootCA.key -aes256 -pkeyopt rsa_keygen_bits:4096
  • openssl genpkey: 这个命令用于生成私钥。genpkey 是 OpenSSL 的一个子命令，专门用于生成私钥。
  • -algorithm RSA: 指定生成的私钥算法是 RSA 算法。RSA 是最常用的公钥加密算法。
  • -out rootCA.key: 指定输出文件名，这里生成的私钥将保存在 rootCA.key 文件中。
  • -aes256: 该参数表示私钥将使用 AES 256 位加密算法进行加密。你将需要一个密码来解密私钥。
  • -pkeyopt rsa_keygen_bits:4096：指定生成的 RSA 密钥长度为 4096 位。

生成根证书

1

openssl req -key rootCA.key -new -x509 -out rootCA.crt -days 3650
  • openssl req: 该命令用于生成证书签署请求（CSR）或者自签名证书。
  • -key rootCA.key: 使用刚才生成的 rootCA.key 私钥来签署证书。
  • -new: 创建一个新的证书请求。
  • -x509: 这告诉 OpenSSL 生成一个自签名证书（而非普通的 CSR）。x509 是常见的证书标准格式。
  • -out rootCA.crt: 该参数指定了生成的证书的输出文件名，即根证书将保存在 rootCA.crt 文件中。
  • -days 3650: 设置证书的有效期为 3650 天（即 10 年）。你可以根据需要修改这个值。

客户生成 CSR（证书签署请求）

openssl req -new -newkey rsa:4096 -nodes -keyout example.com.key -out example.com.csr

# 如果已有私钥，可以使用 —key 来指定
openssl req -new -key example.com.key -out example.com.csr
  • openssl req: 这个命令用于生成证书请求（CSR）。
  • -new: 创建一个新的证书请求。
  • -newkey rsa:2048: 生成一个新的私钥，这里指定使用 RSA 算法和 4096 位的密钥长度。
  • -nodes: 表示不要对生成的私钥加密（即没有密码保护）。如果你希望私钥有密码保护，可以省略这个参数。
  • -keyout example.com.key: 生成的私钥将被保存在 example.com.key 文件中。
  • -out example.com.csr: 生成的证书签署请求（CSR）将被保存在 example.com.csr 文件中。

CA机构签发证书

1

openssl x509 -req -in example.com.csr -CA rootCA.crt -CAkey rootCA.key -CAcreateserial -out example.com.crt -days 365
  • openssl x509: 该命令用于操作 X.509 证书，通常用于签发证书或查看证书信息。
  • -req: 指定输入的是一个证书签署请求（CSR）。这告诉 OpenSSL 你要签署一个证书。
  • -in example.com.csr: 这是输入文件，指向你之前生成的 CSR 文件 example.com.csr。
  • -CA rootCA.crt: 这是证书颁发机构（CA）的根证书。我们使用这个证书来签发新的证书。
  • -CAkey rootCA.key: 这是与根证书配套的私钥，使用这个私钥对新证书进行签署。
  • -CAcreateserial: 如果根证书是第一次签发证书，使用该参数来创建一个序列号文件。这个序列号会存储签发的证书编号。
  • -out example.com.crt: 这是输出文件，签发的证书将保存在 example.com.crt 文件中。
  • -days 365: 设置证书的有效期为 365 天。你可以根据需要调整这个数字。

CA机构撤销证书

撤销证书

1

openssl ca -revoke example.com.crt
  • openssl ca: 该命令用于管理证书颁发和撤销。它可以用来创建证书撤销列表（CRL），更新证书数据库等。
  • -revoke example.com.crt: 这个选项告诉 OpenSSL 撤销 example.com.crt 证书。

生成/更新撤销证书列表（CRL）

1

openssl ca -gencrl -out crl.pem
  • openssl ca: 该命令用于管理证书颁发和撤销。它可以用来创建证书撤销列表（CRL），更新证书数据库等。
  • -gencrl: 这个选项告诉 OpenSSL 生成撤销证书列表（CRL）。
  • -out crl.pem: 指定撤销证书列表的输出文件名，生成的 CRL 会保存在 crl.pem 文件中。

检查撤销证书

1

openssl verify -crl_check -CAfile rootCA.crt example.com.crt
  • rootCA.crt：CA 证书的路径。
  • example.com.crt：要验证的证书路径。
